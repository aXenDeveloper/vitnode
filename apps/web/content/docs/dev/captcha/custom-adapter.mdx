description: Create your own custom captcha adapter for VitNode.

If you want to use captcha in your custom form or somewhere else, follow these steps.

## Usage

<Steps>
<Step>

### Activate captcha in route

```ts title="plugins/{plugin_name}/src/routes/example.ts"
import { buildRoute } from "@vitnode/core/api/lib/route";

export const exampleRoute = buildRoute({
  pluginId: CONFIG_PLUGIN.pluginId,
  route: {
    method: "post",
    description: "Create a new user",
    path: "/sign_up",
    withCaptcha: true, // [!code ++]
  },
  handler: async c => {},
});
```

</Step>
<Step>

### Read the captcha config

```tsx title="apps/web/src/routes/_main/sign-up.tsx"
import { middlewareConfigQueryOptions } from "@vitnode/core/tanstack/auth"; // [!code ++]

export const Route = createFileRoute("/_main/sign-up")({
  loader: async ({ context }) =>
    await context.queryClient.ensureQueryData(middlewareConfigQueryOptions()), // [!code ++]
  component: SignUpRoute,
});

const SignUpRoute = () => {
  const { data } = useSuspenseQuery(middlewareConfigQueryOptions()); // [!code ++]

  return <FormSignUp captcha={data.captcha} />;
};
```

</Step>
<Step>

### Use `useCaptcha` hook

Use the `useCaptcha` hook to handle captcha rendering and validation. Remember to add a `div` with `id="vitnode_captcha"` where you want the captcha widget to appear.

```tsx title="plugins/{plugin_name}/src/views/sign-up/sign-up.tsx"
import { AutoForm } from "@vitnode/core/components/form/auto-form";

export const FormSignUp = ({
  captcha, // [!code ++]
}: {
  captcha: z.infer<typeof routeMiddlewareSchema>["captcha"]; // [!code ++]
}) => {
  // [!code ++]
  const { isReady, getToken, onReset } = useCaptcha(captcha);

  const onSubmit = async () => {
    await mutationApi({
      // ...other values,
      captchaToken: await getToken(), // [!code ++]
    });

    // Handle success or error
    // [!code ++]
    onReset(); // Reset captcha after submission
  };

  return (
    <form onSubmit={onSubmit}>
      {/* Render captcha widget */}
      {/* [!code ++] */}
      <div id="vitnode_captcha" />

      <Button disabled={!isReady}>Submit</Button>
    </form>
  );
};
```

</Step>
<Step>

### Submit form with captcha

```ts title="plugins/{plugin_name}/src/views/sign-up/mutation-api.ts"
import type { z } from "zod";

import { fetcherClient } from "@vitnode/core/lib/fetcher-client";

export const mutationApi = async ({
  captchaToken, // [!code ++]
}: {
  // [!code ++]
  captchaToken: string;
}) => {
  await fetcherClient(usersModuleRef, {
    path: "/test",
    method: "post",
    module: "blog",
    captchaToken, // [!code ++]
  });
};
```

</Step>
