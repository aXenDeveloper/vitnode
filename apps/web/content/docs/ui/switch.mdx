## Usage

import { Tab, Tabs } from 'fumadocs-ui/components/tabs'

```tsx
import { AutoForm } from '@vitnode/core/components/form/auto-form'
import { AutoFormSwitch } from '@vitnode/core/components/form/fields/switch'
import { z } from 'zod'

const formSchema = z.object({
  maintenance: z.boolean().default(false),
})
```

```tsx
<AutoForm
  formSchema={formSchema}
  fields={[
    {
      id: 'maintenance',
      component: (props) => (
        <AutoFormSwitch {...props} label="Maintenance mode" />
      ),
    },
  ]}
/>
```

## States

<Preview name="switch-states" />

- `size="sm"` fits dense lists. The default is `"default"`.
- `disabled` dims it. Add `data-disabled` to the parent [Field](/docs/ui/field)
  to dim the label too.
- `aria-invalid` adds a red ring. Point `aria-describedby` at the error.

```tsx
<Field orientation="horizontal">
  <FieldLabel htmlFor="compact">Compact thread list</FieldLabel>
  <Switch id="compact" size="sm" />
</Field>
```

## Nested fields

`AutoFormSwitch` can reveal the settings it controls in the same card. See
[Auto Form - Nested fields](/docs/ui/auto-form#nested-fields).
