## Installation

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Install package from NPM

import { Tab, Tabs } from 'fumadocs-ui/components/tabs';

<Tabs groupId='package-manager' persist items={['pnpm', 'bun', 'npm']}>

```bash tab="pnpm"
pnpm i @vitnode/blog@canary
```

```bash tab="bun"
bun i @vitnode/blog@canary
```

```bash tab="npm"
npm i @vitnode/blog@canary
```

</Tabs>
</Step>
<Step>

### Connect plugin to config

```ts title="src/vitnode.config.ts"
import { blogPlugin } from '@vitnode/blog/config';
```

```ts title="src/vitnode.config.ts"
export const vitNodeConfig = buildConfig({
  plugins: [blogPlugin()], // [!code highlight]
});
```

</Step>
<Step>

### Connect plugin to API config

```ts title="src/vitnode.api.config.ts"
import { blogApiPlugin } from '@vitnode/blog/config.api';
```

```ts title="src/vitnode.api.config.ts"
export const vitNodeApiConfig = buildApiConfig({
  plugins: [blogApiPlugin()], // [!code highlight]
});
```

</Step>
<Step>

### Run dev server

<Tabs groupId='package-manager' persist items={['pnpm', 'bun', 'npm']}>

```bash tab="pnpm"
pnpm dev
```

```bash tab="bun"
bun dev
```

```bash tab="npm"
npm run dev
```

</Tabs>

</Step>
</Steps>
