In VitNode we're implementing helper to send emails with [React Email](https://react.email/) which allows you to create beautiful responsive emails using React.

## Provider

Before you start sending emails, you need to install the provider. Choose one of the following:

- [Resend](/docs/dev/email/resend),
- [SMTP](/docs/dev/email/smtp).

## Usage

```ts title="test.service.ts"
import { Injectable } from '@nestjs/common';
import { EmailHelperService } from 'vitnode-backend/helpers/email/email.service'; // [!code ++]
import { TestArgs } from './test.dto';

@Injectable()
export class TestService {
  constructor(
    private readonly databaseServices: DatabaseService,
    private readonly mailService: EmailHelperService, // [!code ++]
  ) {}

  show({ email }: TestArgs): string {
    const user = await this.databaseService.db.query.core_users.findFirst({
      where: (table, { eq }) => eq(table.email, email),
    });

    if (!user) {
      throw new NotFoundError('User');
    }

    // [!code word:mailService]
    await this.mailService.send({
      to: user.email,
      subject: 'Test email subject',
      message: 'Test email message',
      previewText: 'Test email preview text',
      user,
    });

    return 'Success!';
  }
}
```

### Internationalization (i18n)

We're created a helper function `getTranslationForEmail(namespace, language)` to translate the email templates. You can use it like this:

```tsx
import React from 'react';
import { Text } from '@react-email/components';
import { getTranslationForEmail } from 'vitnode-backend/helpers/email/email'; // [!code ++]

export const ExampleTemplateEmail = ({
  user,
}: {
  user: {
    language: string; // [!code ++]
    name: string;
  };
}) => {
  const t = getTranslationForEmail('admin.core.email', user.language); // [!code ++]

  return (
    <Text>
      Hi! This is a test for {user.name} - {t('surfix_user')}
    </Text>
  );
};
```

...or in your service.

<Callout>`namespaces` are working the same as in the frontend.</Callout>

### Template Helpers

We're providing some helper functions to make your life easier. Using `this.emailHelpersService.getHelpersForEmail()` you can get the following values like:

- frontend_url,
- site_name,
- site_short_name

<Steps>
<Step>

```tsx title="example.email.tsx"
import React from 'react';
import { Text, Button } from '@react-email/components';
import { getTranslationForEmail } from 'vitnode-backend/helpers/email/email';
import { GetHelpersForEmailType } from 'vitnode-backend/helpers/email/email-helpers.type'; // [!code ++]

export const ExampleTemplateEmail = ({
  user,
  helpers, // [!code ++]
}: {
  helpers: GetHelpersForEmailType; // [!code ++]
  user: {
    language: string;
    name: string;
  };
}) => {
  const t = getTranslationForEmail('admin.core.email', user.language);

  return (
    <Text>
      Hi! This is a test for {user.name} - {t('surfix_user')}
      <Button
        className="bg-primary text-primary-foreground rounded-md px-4 py-2.5 text-sm font-medium"
        href={`${helpers.frontend_url}/test`} // [!code ++]
      >
        {t('button')}
      </Button>
    </Text>
  );
};
```

</Step>
<Step>

Modify the service to inject `EmailHelpersService`.

```ts title="test.service.ts"
import { Injectable, Inject } from '@nestjs/common'; // [!code ++]
import { EmailHelperService } from 'vitnode-backend/helpers/email/email.service';
import type { EmailHelpersServiceType } from 'vitnode-backend/helpers/email/email-helpers.type'; // [!code ++]
import { TestArgs } from './test.dto';
import { ExampleTemplateEmail } from './example.email';

@Injectable()
export class TestService {
  constructor(
    private readonly databaseServices: DatabaseService,
    private readonly mailService: EmailService,
    @Inject('EmailHelpersService') // [!code ++]
    private readonly emailHelpersService: EmailHelperService, // [!code ++]
  ) {}

  async show({ email }: TestArgs): Promise<string> {
    const user = await this.databaseService.db.query.core_users.findFirst({
      where: (table, { eq }) => eq(table.email, email),
    });

    if (!user) {
      throw new NotFoundError('User');
    }

    await this.mailService.send({
      to: user.email,
      subject: 'Test email subject',
      message: SendForgotPasswordTemplateEmail({
        user,
        helpers: await this.emailHelpersService.getHelpersForEmail(), // [!code ++]
      }),
      previewText: 'Test email preview text',
      user,
    });

    return 'Success!';
  }
}
```

## Checking if email is enabled

You can check if email is enabled by using `this.mailService.checkIfEnable()`:

```ts title="test.service.ts"
import { EmailHelperService } from 'vitnode-backend/helpers/email/email.service';

@Injectable()
export class TestService {
  constructor(private readonly mailService: EmailHelperService) {}

  show() {
    if (this.mailService.checkIfEnable()) {
      // Do something
    }
  }
}
```
