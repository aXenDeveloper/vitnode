## Backend

import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>

### Query (DTO)

Create a new file for your DTO. You need to extend the `PaginationQuery` class to show `first`, `last` and `cursor` fields.

```ts title="apps/shared/plugins/{your_plugin_code}/example.dto.ts"
import { IsString, IsOptional } from 'class-validator';
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import { PaginationQuery } from 'vitnode-shared/utils/pagination.dto'; // [!code ++]

export class ShowExampleWelcomeQuery extends PaginationQuery {} // [!code ++]
```

</Step>

<Step>

### Object (DTO)

Our second DTO will be for the return object. You need to extend the `PaginationObj` class and add `edges` field. The `PaginationObj` will add `total_count`, `end_cursor`, `count`, `has_next_page`, `has_previous_page` and `start_cursor` fields.

```ts title="apps/shared/plugins/{your_plugin_code}/example.dto.ts"
import { IsString, IsOptional } from 'class-validator';
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import {
  PaginationObj,
  PaginationQuery,
} from 'vitnode-shared/utils/pagination.dto'; // [!code ++]

export class ShowExampleWelcomeQuery extends PaginationQuery {}

// [!code ++]
export class ExampleWelcomeObj {
  @ApiProperty() // [!code ++]
  name: string; // [!code ++]
  // [!code ++]
}

// [!code ++]
export class ShowExampleWelcomeObj extends PaginationObj {
  @ApiProperty({ type: [ExampleWelcomeObj] }) // [!code ++]
  edges: ExampleWelcomeObj[]; // [!code ++]
  // [!code ++]
}
```

</Step>

<Step>

### Import DTO to Service

Now you need to import the DTO to your service.

```ts title="apps/backend/src/plugins/{your_plugin}/admin/services/example.service.ts"
import { Injectable } from '@nestjs/common';
import {
  ShowExampleWelcomeQuery,
  ShowExampleWelcomeObj,
} from 'shared/plugins/{your_plugin_code}/example.dto';

@Injectable()
export class ExampleWelcomeService {
  async show({
    first,
    last,
    cursor,
  }: ShowExampleWelcomeQuery): Promise<ShowExampleWelcomeObj> {}
}
```

</Step>

<Step>

### Query to Database

In your service, you need to initialize the database using `constructor()` from the `DatabaseService` and use the `paginationCursor` method to get data with cursor pagination. As an example, we will use the `core_languages` table.

```ts title="apps/backend/src/plugins/{your_plugin}/admin/services/example.service.ts"
import { Injectable } from '@nestjs/common';
import { DatabaseService } from '@/database/database.service'; // [!code ++]
import { SortDirectionEnum } from 'vitnode-shared/utils/pagination.enum'; // [!code ++]
import {
  ShowExampleWelcomeQuery,
  ShowExampleWelcomeObj,
} from 'shared/plugins/{your_plugin_code}/example.dto';
import { core_languages } from '@/database/schema/langs'; // [!code ++]

@Injectable()
export class ExampleWelcomeService {
  constructor(private readonly databaseService: DatabaseService) {} // [!code ++]

  async show({
    first,
    last,
    cursor,
  }: ShowExampleWelcomeQuery): Promise<ShowExampleWelcomeObj> {
    // [!code ++]
    const pagination = await this.databaseService.paginationCursor({
      cursor, // [!code ++]
      database: core_languages, // [!code ++]
      first, // [!code ++]
      last, // [!code ++]
      // [!code ++]
      defaultSortBy: {
        direction: SortDirectionEnum.desc, // [!code ++]
        column: 'updated_at', // [!code ++]
      }, // [!code ++]
      // [!code ++]
      query: async args =>
        await this.databaseService.db.query.core_languages.findMany(args),
    });

    return pagination; // [!code ++]
  }
}
```

</Step>
</Steps>

## Advanced Usage

### Where Arguments

If you want to use `where` argument you can pass it to the `findMany()` method using `and()` method from `drizzle-orm`.

```ts title="apps/backend/src/plugins/{your_plugin}/admin/services/example.service.ts"
import { ilike } from 'drizzle-orm'; // [!code ++]
import { Injectable } from '@nestjs/common';
import { DatabaseService } from '@/database/database.service';
import { SortDirectionEnum } from 'vitnode-shared/utils/pagination.enum';
import {
  ShowExampleWelcomeQuery,
  ShowExampleWelcomeObj,
} from 'shared/plugins/{your_plugin_code}/example.dto';
import { core_languages } from '@/database/schema/langs';

@Injectable()
export class ExampleWelcomeService {
  constructor(private readonly databaseService: DatabaseService) {}

  async show({
    first,
    last,
    cursor,
  }: ShowExampleWelcomeQuery): Promise<ShowExampleWelcomeObj> {
    const search = 'english'; // [!code ++]
    const where = ilike(core_languages.name, `%${search}%`); // [!code ++]

    const pagination = await this.databaseService.paginationCursor({
      cursor,
      database: core_languages,
      first,
      last,
      where, // [!code ++]
      defaultSortBy: {
        direction: SortDirectionEnum.desc,
        column: 'updated_at',
      },
      query: async args =>
        await this.databaseService.db.query.core_languages.findMany(args),
    });

    return pagination;
  }
}
```

### Custom Query

If you want to use a custom query, you can pass it to the `query` argument.

```ts title="apps/backend/src/plugins/{your_plugin}/admin/services/example.service.ts"
const pagination = await this.databaseService.paginationCursor({
  cursor,
  database: core_languages,
  first,
  last,
  defaultSortBy: {
    direction: SortDirectionEnum.desc,
    column: "updated_at"
  },
  query: async (args) => await this.databaseService.db.query.core_languages.findMany(args) // [!code --]
  // [!code ++]
  query: async (args) =>
  // [!code ++]
    await this.databaseService.db.query.core_languages.findMany({
      ...args, // [!code ++]
      columns: { // [!code ++]
        id: true, // [!code ++]
        name: true, // [!code ++]
        updated_at: true // [!code ++]
        // [!code ++]
      }
      // [!code ++]
    })
});
```

<Callout type="warn" emoji="⚠️" title="Where argument in query">
  Do not change `where` argument in the `query` method. It will couse an issue
  with `total_count` field.
</Callout>

### Primary cursor

As default `primaryCursor` is set to `id` column. If you want to change it, you can pass the `primaryCursor` argument.

```ts title="apps/backend/src/plugins/{your_plugin}/admin/services/example.service.ts"
const pagination = await this.databaseService.paginationCursor({
  cursor,
  database: core_languages,
  first,
  last,
  primaryCursor: 'id_something', // [!code ++]
  defaultSortBy: {
    direction: SortDirectionEnum.desc,
    column: 'updated_at',
  },
  query: async args =>
    await this.databaseService.db.query.core_languages.findMany(args),
});
```
