## Usage

import { Tab, Tabs } from 'fumadocs-ui/components/tabs';

<Tabs items={['Auto Form', 'Manual']}>
  <Tab value="Auto Form">
```ts
import { z } from 'zod';
import { AutoForm } from 'vitnode-frontend/form/auto-form';
import { AutoFormColorPicker } from 'vitnode-frontend/form/fields/color-picker';
```

```ts
const formSchema = z.object({
  color_primary: z.string(),
});
```

```tsx
<AutoForm
  formSchema={formSchema}
  fields={[
    {
      id: 'color_primary',
      component: AutoFormColorPicker, // [!code highlight]
    },
  ]}
/>
```

  </Tab>

  <Tab value="Manual">
```tsx
import { ColorPicker } from 'vitnode-frontend/components/ui/color-picker';
```

```tsx
<ColorPicker onChange={onChange} value={value} />
```

  </Tab>
</Tabs>
