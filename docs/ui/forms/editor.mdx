## Usage

import { Tab, Tabs } from 'fumadocs-ui/components/tabs';

<Tabs items={['Auto Form', 'Manual']}>
  <Tab value="Auto Form">
```ts
import { z } from 'zod';
import { AutoForm } from 'vitnode-frontend/form/auto-form';
import { AutoFormEditor } from 'vitnode-frontend/form/fields/editor';
import { zodLanguageInput } from 'vitnode-frontend/helpers/zod';
```

```ts
const formSchema = z.object({
  content: zodLanguageInput, // [!code highlight]
});
```

```tsx
<AutoForm
  formSchema={formSchema}
  fields={[
    {
      id: 'content',
      component: AutoFormEditor,
    },
  ]}
/>
```

  </Tab>

  <Tab value="Manual">

```tsx
'use client';

import React from 'react';
import { Editor } from 'vitnode-frontend/components/editor/editor';
import { StringLanguage } from 'vitnode-shared/string-language.dto';

export const EditorDemo = () => {
  const [value, setValue] = React.useState<StringLanguage[]>([]);

  return <Editor onChange={setValue} value={value} />;
};
```

  </Tab>
</Tabs>

## Upload Files

Our editor allows you to upload files. To enable this feature, you need to pass the `allowUploadFiles` prop with the plugin and folder name.

<Tabs items={['Auto Form', 'Manual']}>
  <Tab value="Auto Form">
```ts
import { z } from 'zod';
import { AutoForm } from 'vitnode-frontend/form/auto-form';
import { AutoFormEditor } from 'vitnode-frontend/form/fields/editor';
import { zodLanguageInput } from 'vitnode-frontend/helpers/zod';
```

```ts
const formSchema = z.object({
  content: zodLanguageInput, // [!code highlight]
});
```

```tsx
<AutoForm
  formSchema={formSchema}
  fields={[
    {
      id: 'content',
      component: props => (
        <AutoFormEditor
          {...props}
          allowUploadFiles={{
            plugin_code: '{your_plugin_name}',
            folder: '{your_folder_to_save_files}',
          }}
        />
      ),
    },
  ]}
/>
```

  </Tab>

  <Tab value="Manual">

```tsx
<Editor
  allowUploadFiles={{
    plugin_code: '{your_plugin_name}',
    folder: '{your_folder_to_save_files}',
  }}
  onChange={onChange}
  value={value}
/>
```

  </Tab>
</Tabs>

## Read only mode

To display content form editor you have to use `ReadOnlyEditor` component.

```tsx
import { ReadOnlyEditor } from 'vitnode-frontend/components/editor/read-only';
```

```tsx
<ReadOnlyEditor value={value} />
```
